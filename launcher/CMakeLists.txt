cmake_minimum_required(VERSION 3.28)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(obs-quick-record-native LANGUAGES CXX RC)
endif()
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The native assistant requires Windows x64 MSVC")
endif()
enable_language(RC)
add_executable(obs-quick-record-native WIN32 native-main.cpp native-support.cpp native-check.cpp native-launcher.rc)
target_compile_features(obs-quick-record-native PRIVATE cxx_std_20)
target_include_directories(obs-quick-record-native PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(obs-quick-record-native PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(obs-quick-record-native PRIVATE /W4 /utf-8 /permissive- /EHsc)
target_link_libraries(obs-quick-record-native PRIVATE windowsapp bcrypt ole32 shell32 advapi32 user32 psapi uuid)
set_target_properties(obs-quick-record-native PROPERTIES AUTOMOC OFF MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CheckIPOSupported)
check_ipo_supported(RESULT nativeIPO LANGUAGES CXX)
if(nativeIPO)
  set_property(TARGET obs-quick-record-native PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  include(CTest)
  add_test(NAME native-launcher-check COMMAND obs-quick-record-native --check)
  set_tests_properties(native-launcher-check PROPERTIES TIMEOUT 60)
  find_program(POWERSHELL_EXECUTABLE NAMES pwsh powershell REQUIRED)
  add_test(NAME native-discovery-check COMMAND "${POWERSHELL_EXECUTABLE}" -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/../tests/native-discovery-check.ps1" -NativeExe $<TARGET_FILE:obs-quick-record-native>)
  set_tests_properties(native-discovery-check PROPERTIES TIMEOUT 90)
endif()
